//@version=5
strategy("Institutional Session ORB & Multi-Stage Expansion [NSE Quant Studio]", 
         shorttitle="Institutional ORB v3 [5m]",
         overlay=true, 
         initial_capital=500000, 
         default_qty_type=strategy.fixed, 
         default_qty_value=75, 
         currency=currency.INR,
         commission_type=strategy.commission.cash_per_order, 
         commission_value=0.0)

// =============================================================================
// INPUTS & CONFIGURATION
// =============================================================================
grp_orb   = "1. Opening Range Parameters"
or_len    = input.int(15, "Opening Range Duration (Minutes)", options=[15, 30], group=grp_orb)

grp_st    = "2. Supertrend Parameters"
st_period = input.int(10, "Supertrend ATR Period", group=grp_st)
st_mult   = input.float(3.0, "Supertrend Multiplier", step=0.1, group=grp_st)

grp_filt  = "3. Trend & Expansion Filters"
adx_len   = input.int(14, "ADX Period", group=grp_filt)
adx_thres = input.float(20.0, "Minimum ADX Strength", step=1.0, group=grp_filt)
use_vwap  = input.bool(true, "Require VWAP Slope Filter", group=grp_filt)

grp_risk  = "4. Risk & Ratchet Engine"
target_r  = input.float(3.0, "Profit Target (R-Multiple)", step=0.5, group=grp_risk)
be_trig_r = input.float(1.2, "Breakeven Trigger (R-Multiple)", step=0.1, group=grp_risk)
be_lock_r = input.float(0.2, "Breakeven Lock Profit (R-Multiple)", step=0.05, group=grp_risk)
use_trail = input.bool(true, "Dynamic Supertrend Trailing Stop", group=grp_risk)

grp_sess  = "5. Institutional Trading Sessions (IST)"
morn_sess = input.session("0930-1115", "Morning Institutional Window", group=grp_sess)
lunch_nop = input.session("1115-1315", "Lunch Chop No-Trade Zone", group=grp_sess)
aft_sess  = input.session("1315-1445", "Afternoon Institutional Trend", group=grp_sess)
mis_exit  = input.session("1515-1530", "Mandatory MIS Square-Off Time", group=grp_sess)

// =============================================================================
// 1. OPENING RANGE ENGINE (09:15 to 09:30 or 09:45 IST)
// =============================================================================
var float or_high = na
var float or_low  = na
var float or_mid  = na
var int   morn_trades_today = 0
var int   aft_trades_today  = 0

is_new_day = ta.change(time("D")) != 0

if is_new_day
    or_high := na
    or_low  := na
    or_mid  := na
    morn_trades_today := 0
    aft_trades_today  := 0

t_int = hour(time, "Asia/Kolkata") * 100 + minute(time, "Asia/Kolkata")
is_or_calc_bar = (or_len == 15 and t_int >= 915 and t_int < 930) or (or_len == 30 and t_int >= 915 and t_int < 945)

if is_or_calc_bar
    or_high := na(or_high) ? high : math.max(or_high, high)
    or_low  := na(or_low) ? low : math.min(or_low, low)
    or_mid  := (or_high + or_low) / 2.0

plot(not is_or_calc_bar and not na(or_high) ? or_high : na, "OR High", color=color.green, linewidth=2, style=plot.style_linebr)
plot(not is_or_calc_bar and not na(or_low) ? or_low : na, "OR Low", color=color.red, linewidth=2, style=plot.style_linebr)
plot(not is_or_calc_bar and not na(or_mid) ? or_mid : na, "OR Midpoint Stop", color=color.orange, linewidth=1, style=plot.style_linebr)

// =============================================================================
// 2. INDICATOR ENGINE: SUPERTREND, ADX, AND INTRADAY VWAP
// =============================================================================
[st_value, st_direction] = ta.supertrend(st_mult, st_period)
st_bull = st_direction < 0
st_bear = st_direction > 0

[plus_di, minus_di, adx_val] = ta.dmi(adx_len, adx_len)
adx_expansion = adx_val >= adx_thres

cur_vwap = ta.vwap(hlc3)
vwap_slope_up = cur_vwap >= cur_vwap[1]
vwap_slope_dn = cur_vwap <= cur_vwap[1]

// =============================================================================
// 3. SESSION FILTERS & TIME CONTROL
// =============================================================================
in_morning_win   = not na(time(timeframe.period, morn_sess + ":23456", "Asia/Kolkata")) and ((or_len == 15 and t_int >= 930) or (or_len == 30 and t_int >= 945))
in_lunch_zone    = not na(time(timeframe.period, lunch_nop + ":23456", "Asia/Kolkata"))
in_afternoon_win = not na(time(timeframe.period, aft_sess  + ":23456", "Asia/Kolkata"))
is_mis_exit_bar  = not na(time(timeframe.period, mis_exit  + ":23456", "Asia/Kolkata")) or t_int >= 1515

bgcolor(in_morning_win   ? color.new(color.green, 93) : 
        in_lunch_zone    ? color.new(color.purple, 93) : 
        in_afternoon_win ? color.new(color.blue, 93) : na)

// =============================================================================
// 4. POSITION RISK & RATCHET STATE
// =============================================================================
var float entry_price  = na
var float active_stop  = na
var float init_risk    = na
var float target_price = na
var bool  be_locked    = false

if strategy.position_size == 0
    entry_price  := na
    active_stop  := na
    init_risk    := na
    target_price := na
    be_locked    := false

if strategy.position_size > 0
    if not be_locked and high >= entry_price + init_risk * be_trig_r
        active_stop := math.max(active_stop, entry_price + init_risk * be_lock_r)
        be_locked   := true
    if use_trail and st_bull and st_value > active_stop
        active_stop := math.max(active_stop, st_value)

if strategy.position_size < 0
    if not be_locked and low <= entry_price - init_risk * be_trig_r
        active_stop := math.min(active_stop, entry_price - init_risk * be_lock_r)
        be_locked   := true
    if use_trail and st_bear and st_value < active_stop
        active_stop := math.min(active_stop, st_value)

// =============================================================================
// 5. ENTRY EXECUTION LOGIC
// =============================================================================
can_enter_morn = in_morning_win and (morn_trades_today < 1)
can_enter_aft  = in_afternoon_win and (aft_trades_today < 1)
can_enter      = (can_enter_morn or can_enter_aft) and strategy.position_size == 0 and not na(or_high)

long_setup  = can_enter and (close > or_high) and st_bull and adx_expansion and (not use_vwap or (close > cur_vwap and vwap_slope_up))
short_setup = can_enter and (close < or_low)  and st_bear and adx_expansion and (not use_vwap or (close < cur_vwap and vwap_slope_dn))

if long_setup
    entry_price  := close
    active_stop  := or_mid
    init_risk    := math.max(entry_price - active_stop, 10.0)
    target_price := entry_price + target_r * init_risk
    be_locked    := false
    strategy.entry("Long_ORB", strategy.long)
    if in_morning_win
        morn_trades_today += 1
    else
        aft_trades_today += 1

if short_setup
    entry_price  := close
    active_stop  := or_mid
    init_risk    := math.max(active_stop - entry_price, 10.0)
    target_price := entry_price - target_r * init_risk
    be_locked    := false
    strategy.entry("Short_ORB", strategy.short)
    if in_morning_win
        morn_trades_today += 1
    else
        aft_trades_today += 1

// =============================================================================
// 6. EXIT EXECUTION
// =============================================================================
if is_mis_exit_bar and strategy.position_size != 0
    strategy.close_all(comment="MIS Auto-Squareoff")

if strategy.position_size > 0
    strategy.exit("Exit_Long", "Long_ORB", stop=active_stop, limit=target_price, comment_loss="Stop/BE", comment_profit="Target")
    if st_bear
        strategy.close("Long_ORB", comment="ST Flip Exit")

if strategy.position_size < 0
    strategy.exit("Exit_Short", "Short_ORB", stop=active_stop, limit=target_price, comment_loss="Stop/BE", comment_profit="Target")
    if st_bull
        strategy.close("Short_ORB", comment="ST Flip Exit")
